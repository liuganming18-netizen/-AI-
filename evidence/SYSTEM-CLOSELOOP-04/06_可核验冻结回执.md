## 4. 关键数据库表结构 (DDL 元数据)

```sql
-- 1. 委托订单表 (paper_order)
CREATE TABLE paper_order(
  order_id INTEGER PRIMARY KEY AUTOINCREMENT, account_id TEXT, event_uid TEXT UNIQUE,
  symbol TEXT, side TEXT, qty REAL, price REAL, fee REAL, ts INTEGER,
  status TEXT, reject_reason TEXT, fill_price REAL, notional REAL, bar_open_time INTEGER,
  model_version TEXT, event_time INTEGER, requested_at INTEGER, market_source TEXT,
  first_request_ms INTEGER, epoch_ms INTEGER, voided INTEGER DEFAULT 0, void_reason TEXT);

-- 2. 真实成交表 (paper_fill)
CREATE TABLE paper_fill(
  fill_id INTEGER PRIMARY KEY AUTOINCREMENT, account_id TEXT NOT NULL,
  order_id INTEGER, position_id INTEGER, signal_uid TEXT, symbol TEXT NOT NULL,
  action TEXT NOT NULL, qty REAL NOT NULL, raw_price REAL NOT NULL, fill_price REAL NOT NULL,
  notional REAL NOT NULL, fee REAL NOT NULL, slippage_cost REAL NOT NULL,
  bar_open_time INTEGER, bar_close_time INTEGER, fill_ts INTEGER NOT NULL,
  model_version TEXT, event_time INTEGER, available_at INTEGER, requested_at INTEGER,
  market_source TEXT, first_request_ms INTEGER, epoch_ms INTEGER, voided INTEGER DEFAULT 0, void_reason TEXT);

-- 3. 仓位状态表 (paper_position)
CREATE TABLE paper_position(
  position_id INTEGER PRIMARY KEY AUTOINCREMENT, account_id TEXT, symbol TEXT,
  qty REAL, entry_price REAL, entry_ts INTEGER, exit_ts INTEGER, exit_reason TEXT,
  status TEXT, exit_price REAL, entry_fee REAL, exit_fee REAL, funding_paid REAL DEFAULT 0,
  signal_uid TEXT, arm TEXT, stop_price REAL, atr_at_entry REAL, model_version TEXT,
  event_time INTEGER, available_at INTEGER, requested_at INTEGER, earliest_fill_ms INTEGER,
  market_source TEXT, origin TEXT, first_request_ms INTEGER, epoch_ms INTEGER,
  voided INTEGER DEFAULT 0, void_reason TEXT);

-- 4. FT-LOW4H 止损上移轨迹表 (paper_stop_history)
CREATE TABLE paper_stop_history(
  id INTEGER PRIMARY KEY AUTOINCREMENT, account_id TEXT NOT NULL, position_id INTEGER NOT NULL,
  old_stop REAL, new_stop REAL NOT NULL, reason TEXT NOT NULL,
  ref_bar_open_ms INTEGER, available_at INTEGER, decision_at INTEGER NOT NULL,
  effective_at INTEGER NOT NULL, created_at INTEGER NOT NULL);
```
